# ── COPIAS ANTIGAS DO CODIGO ────────────────────────────────────────────────
# Em 20/08/2026 descobrimos que "catalogo-old-20260719.php" continuava servindo
# as avaliacoes SEM TOKEN, mesmo depois de o catalogo.php ter sido fechado.
# Fechar a porta da frente nao adianta enquanto a copia antiga continua no ar
# com o codigo velho — e nenhuma correcao futura chega nela.
# O certo e APAGAR esses arquivos do servidor. Esta regra e a rede enquanto isso.
<FilesMatch "-old-[0-9]{8}\.php$">
    Require all denied
</FilesMatch>

# Arquivos de diagnostico do PHP jamais devem ficar acessiveis: revelam versao,
# extensoes, caminho absoluto do site e variaveis de ambiente.
<FilesMatch "^(phpinfo|info|test|teste)[-_a-z0-9]*\.php$">
    Require all denied
</FilesMatch>

# Helpers: so existem para ser incluidos por outros arquivos.
<FilesMatch "^_.*\.php$">
    Require all denied
</FilesMatch>

# Nenhum destes formatos deve ser servido a partir da pasta da API.
<FilesMatch "\.(json|sql|bak|zip|tar|gz|old|orig|save|swp|yml|yaml)$">
    Require all denied
</FilesMatch>
